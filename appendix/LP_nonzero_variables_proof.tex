\section{Nonzero Variables in the LP solution}

\begin{lemma}[Nonzero Variables in the LP Solution]
\label{lem:lp_nonzero_variables}
The LP relaxation of \Cref{eq:bp-ilp} has an optimal
vertex solution with at most $m$ nonzero variables.
\end{lemma}

\begin{proof}
    Let $A$ be the configuration matrix with $m$ rows.
    The feasible region is
    \[
        P=\{\mathbf{a}\geq\mathbf{0}:A\mathbf{a}\geq\mathbf{n}\}.
    \]
    Since $P$ contains no lines and the LP has a finite
    optimum, an optimal vertex solution $\mathbf{a}^*$ exists.

    Suppose its support $S=\{j:a_j^*>0\}$ has more than
    $m$ elements. The corresponding columns of $A$ are
    linearly dependent, so there is a nonzero vector
    $\mathbf{d}$ satisfying
    \[
        A\mathbf{d}=\mathbf{0},
        \qquad d_j=0\quad\text{for }j\notin S.
    \]
    Since $a_j^*>0$ on $S$, a sufficiently small
    $\eps>0$ ensures $\mathbf{a}^*\pm\eps\mathbf{d}\geq\mathbf{0}$.
    Moreover,
    \[
        A(\mathbf{a}^*\pm\eps\mathbf{d})
        =A\mathbf{a}^*\geq\mathbf{n}.
    \]
    Thus, $\mathbf{a}^*$ is the midpoint of two distinct
    feasible solutions:
    \[
        \mathbf{a}^*
        =\frac{1}{2}(\mathbf{a}^*+\eps\mathbf{d})
        +\frac{1}{2}(\mathbf{a}^*-\eps\mathbf{d}).
    \]
    This contradicts its being a vertex. Hence,
    $\mathbf{a}^*$ has at most $m$ nonzero components.
\end{proof}
